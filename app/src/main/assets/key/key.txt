# "Work through it": the questions, in the order they are asked, and which first-aid pages
# each answer points to or rules out. Ids only; every word shown lives in strings.xml
# (key_q_<question>, key_a_<question>_<answer>, key_n_<question>), so this file never needs
# translating, and each line can be checked against its page.
#
#   stage <name>                     the response chain: scene threats call look treat watch
#   ask <id> [options] : <answers>   a question, in chain order
#       label=X        the small letter in the margin (X, A, B, C, D, E)
#       tag=AVPU       a small word at the right of the question
#       many           pick all that apply; nothing picked says nothing. Never in a rules line.
#       after=<when>   shown only once <when> holds (a follow-up)
#       settle         asked only to settle a page still to check
#   grid <id> places <p…> findings <f…>   head to toe: one tap per finding at each place
#
#   page <id> [danger]
#     points <when>    any one clause puts the page under Fits
#     rules <when>     danger pages only: any one clause, every part of it a sure answer,
#                      rules the page out. "Not sure" is no answer and rules out nothing.
#
#   <when>: clauses split by " | ", each a list of q=a split by " & ". A "*" in a grid answer
#   stands for any place or finding. Two answers come from the vitals rather than a question:
#   pulse=fast (over 100 a minute) and vskin=pale_cool (pale or grey, cool or clammy).
#
# Every single-answer question also offers "not sure" (not_sure).

stage scene
ask safe : yes no
ask what many : fall water cold heat bite_sting lightning swallowed by_itself dont_know
ask count : one more

stage threats
ask bleeding label=X : none some heavy
ask avpu label=A tag=AVPU : alert voice pain none
ask breathing label=B : normal struggling gasping
ask skin label=C : yes no
ask move label=D : yes no
ask head_hit label=D : yes no
ask confused label=D : yes no
ask cold label=E : yes no
ask hot label=E : yes no

stage call
ask plan : called walk carry stay

stage look
ask seen many : hurt_limb pain hard_breathe chest_pain choking wound cold_shiver overheated face_speech bite_sting rash_swelling burn fit numb_white thirsty water high_up swallowed diabetic blister tick
ask deformed after=seen=hurt_limb : yes no
ask use_limb after=seen=hurt_limb : yes no
ask bite_kind after=seen=bite_sting|what=bite_sting : snake insect tick other
ask face after=seen=face_speech|confused=yes settle : droops even
ask arm after=seen=face_speech|confused=yes settle : drifts level
ask speech after=seen=face_speech|confused=yes settle : slurred clear
ask chest settle : yes no
ask diabetic settle : yes no
grid body places head neck chest belly back pelvis arms legs findings hurt bleeding deformed numb
ask you : yes no

# Pages, in the order the First aid list shows them.

# Reached from the Scene stage as well.
page scene
  points safe=no | count=more

# The red flags: what the Call stage lists as reasons to call. Reached from that stage.
page call
  points safe=no | count=more | bleeding=heavy | avpu=voice | avpu=pain | avpu=none | breathing=struggling | breathing=gasping | skin=yes | move=no | head_hit=yes | confused=yes | seen=chest_pain | chest=yes | seen=choking | seen=hard_breathe | seen=face_speech | face=droops | arm=drifts | speech=slurred | seen=fit | seen=diabetic | seen=swallowed | what=swallowed | what=lightning | what=water | seen=water | bite_kind=snake | deformed=yes | use_limb=no | body=*.deformed | body=*.numb | seen=overheated & confused=yes | seen=cold_shiver & confused=yes | pulse=fast & vskin=pale_cool

page cpr danger
  points breathing=gasping
  rules breathing=normal | breathing=struggling

page choking danger
  points seen=choking
  rules breathing=normal

page bleeding danger
  points bleeding=heavy
  rules bleeding=none | bleeding=some

page drowning
  points what=water | seen=water

page anaphylaxis danger
  points seen=rash_swelling & breathing=struggling | seen=rash_swelling & seen=hard_breathe | seen=rash_swelling & skin=yes | bite_kind=insect & breathing=struggling | bite_kind=insect & seen=hard_breathe | bite_kind=insect & skin=yes | what=bite_sting & breathing=struggling | what=bite_sting & seen=hard_breathe
  rules breathing=normal & skin=no

page breathing danger

  rules breathing=normal

page chest danger

# Patient assessment: the questions, in the order they are asked, and which first-aid pages
# each answer points to or rules out. Ids only; every word shown lives in strings.xml
# (key_q_<question>, key_a_<question>_<answer>, key_n_<question>), so this file never needs
# translating, and each line can be checked against its page.
#
#   stage <name>                     the Patient Assessment System, in order:
#                                    scene (size-up) primary call secondary treat monitor
#   part <name>                      a part of the secondary assessment: chief, exam, sample, focused
#   ask <id> [options] : <answers>   a question, in assessment order
#       label=X        the small letter in the margin (X, A, B, C, D, E; F, A, S for the stroke check)
#       tag=AVPU       a small word at the right of the question
#       many           pick all that apply; nothing picked says nothing. Never in a rules line.
#       after=<when>   shown only once <when> holds (a follow-up)
#       settle         asked only to settle a page still to check
#       decision       the rescuer's own decision, not a finding: offers "Not decided yet" in
#                      place of "not sure", and no page reads it
#   grid <id> places <p…> findings <f…>   head-to-toe exam: one tap per finding at each place.
#                      A finding written f:p,q is offered only at those places.
#
#   page <id> [danger]
#     points <when>    any one clause puts the page under the problem list
#     rules <when>     danger pages only: any one clause, every part of it a sure answer,
#                      rules the page out. "Not sure" is no answer and rules out nothing.
#
#   <when>: clauses split by " | ", each a list of q=a split by " & ". A "*" in a grid answer
#   stands for any place or finding. Some answers come from the latest vital signs rather than
#   a question: pulse=fast (HR over 100 a minute), pulse=normal (HR 100 or under),
#   resp=fast (RR over 30 a minute), vskin=pale_cool (pale or grey, cool or clammy) and
#   vskin=blue (blue lips). Until the vital signs are taken, none of them is an answer.
#
# Every single-answer question that is not a decision also offers "not sure" (not_sure).

stage scene
ask safe : yes no
ask what many : fall water cold heat burn bite_sting lightning swallowed by_itself dont_know
ask storm after=what=dont_know|what=by_itself|what=fall : yes no
ask count : one more

# Responsiveness first, then X-ABCDE.
stage primary
ask avpu tag=AVPU : alert voice pain none
ask bleeding label=X : none some heavy
ask airway label=A : clear noisy blocked
ask breathing label=B : normal struggling gasping
ask skin label=C : yes no
ask head_hit label=D : yes no
ask move label=D : yes no
ask confused label=D : yes no
ask cold label=E : yes no
ask hot label=E : yes no

stage call
ask called decision : called sent not_yet

stage secondary
part chief
ask seen many : hurt_limb pain hard_breathe chest_pain choking wound cold_shiver overheated face_speech bite_sting rash_swelling lips_tongue burn fit numb_white thirsty vomit_diarrhea water high_up swallowed diabetic blister tick
ask deformed after=seen=hurt_limb : yes no
ask tender after=seen=hurt_limb : yes no
ask use_limb after=seen=hurt_limb : yes no
ask bite_kind after=seen=bite_sting|what=bite_sting : snake insect tick animal other
ask burn_kind many after=seen=burn|what=burn : heat electric chemical smoke
ask burn_serious after=seen=burn|what=burn : yes no
ask frozen after=seen=numb_white : yes no
part exam
grid body places head neck chest belly back pelvis arms legs findings deformed bleeding hurt swelling burned numb:head,arms,legs
part sample
ask you : yes no
ask allergy_known label=A : yes no
ask diabetic label=P : yes no
ask seizure_hx after=avpu=voice|avpu=pain|avpu=none|confused=yes : yes no
ask altitude label=E settle : yes no
part focused
ask face label=F after=seen=face_speech|confused=yes|avpu=voice settle : droops even
ask arm label=A after=seen=face_speech|confused=yes|avpu=voice settle : drifts level
ask speech label=S after=seen=face_speech|confused=yes|avpu=voice settle : slurred clear
ask befast after=seen=face_speech|confused=yes|avpu=voice|avpu=pain settle : yes no
ask walk after=altitude=yes|seen=high_up settle : straight stumbles
ask chest after=seen=pain|seen=hard_breathe|breathing=struggling|skin=yes settle : yes no

stage treat
ask plan decision : stay walk carry rescue

# CPR needs both: unresponsive, and not breathing normally. Labored breathing rules it out
# only while they still respond to you.
page cpr danger
  points breathing=gasping | avpu=none & airway=blocked
  rules breathing=normal | breathing=struggling & avpu=alert | breathing=struggling & avpu=voice

page choking danger
  points seen=choking | airway=blocked
  rules airway=clear & breathing=normal

page bleeding danger
  points bleeding=heavy | bleeding=some & pulse=fast & vskin=pale_cool | body=*.bleeding & pulse=fast & vskin=pale_cool | seen=wound & pulse=fast & vskin=pale_cool
  rules bleeding=none | bleeding=some
